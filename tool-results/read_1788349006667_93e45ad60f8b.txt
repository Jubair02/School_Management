    72→const studentFormSchema = z.object({
    73→  name: z.string().min(2, "Name is required"),
    74→  email: z.email("Enter a valid email address"),
    75→  password: z.string().min(6, "Minimum 6 characters").optional().or(z.literal("")),
    76→  phone: z.string().optional().or(z.literal("")),
    77→  dateOfBirth: z.string().optional(),
    78→  gender: z.enum(["MALE", "FEMALE", "OTHER"]).optional(),
    79→  address: z.string().optional(),
    80→  classId: z.string().optional(),
    81→  sectionId: z.string().optional(),
    82→  parentId: z.string().optional(),
    83→  rollNumber: z.string().optional(),
    84→});
    85→type StudentFormValues = z.infer<typeof studentFormSchema>;
    86→
    87→function cleanPayload(values: StudentFormValues): StudentCreateInput {
    88→  const str = (v: string | undefined) => {
    89→    const t = v?.trim();
    90→    return t ? t : undefined;
    91→  };
    92→  const out: StudentCreateInput = { name: values.name.trim(), email: values.email.trim() };
    93→  const password = str(values.password);
    94→  if (password) out.password = password;
    95→  const phone = str(values.phone);
    96→  if (phone) out.phone = phone;
    97→  if (values.dateOfBirth) out.dateOfBirth = values.dateOfBirth;
    98→  if (values.gender) out.gender = values.gender as Gender;
    99→  const address = str(values.address);
   100→  if (address) out.address = address;
   101→  if (values.classId) out.classId = values.classId;
   102→  if (values.sectionId) out.sectionId = values.sectionId;
   103→  if (values.parentId) out.parentId = values.parentId;
   104→  const roll = str(values.rollNumber);
   105→  if (roll) out.rollNumber = roll;
   106→  return out;
   107→}
